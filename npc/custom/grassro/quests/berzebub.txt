//===== GrassRO Old Woman =======================================
//= Unified repeatable Berzebub Card creation/restoration, v4.0
//= 39999 Torn Berzebub Card (10%)
//= 40000 Partially Restored Berzebub Card (20%)
//= 4145 Original Beelzebub Card (30%)
//=
//= All recipes repeatable. No character/account completion flags.
//= No bind commands. Item trade permissions are controlled by item DB.
//= Previous-stage cards must be loose in inventory and are consumed.
//= #GR_BerzTxn is a persistent fail-closed recovery marker:
//=   1 = Torn creation, 2 = 20% upgrade, 3 = 30% upgrade.
//= Interrupted exchanges are NOT atomic. GM must reconcile manually.
//= Requires F_GrassRO_ItemCreationAnnounce in scripts.conf.
//==============================================================
prontera,165,178,8	script	Old Woman#GR_Berz	103,{
	mes "[Old Woman]";
	if (#GR_BerzTxn) {
		mes "A previous card exchange on this account needs manual review.";
		mes "Please contact a GM and report transaction stage " + #GR_BerzTxn + ".";
		close;
	}

	mes "I can create or restore Berzebub Cards, provided you bring the required materials.";
	mes "Each restoration consumes the previous card. All three services are repeatable.";
	next;

	switch (select("Create Torn Card (10%):Restore to 20%:Restore to original 30%:Learn about the cards:Leave")) {
	case 1: goto L_Torn;
	case 2: goto L_Restored;
	case 3: goto L_Original;
	case 4: goto L_Info;
	default: close;
	}

L_Info:
	mes "[Old Woman]";
	mes "Torn Berzebub Card: 10% cast-time reduction.";
	mes "Partially Restored Berzebub Card: 20% cast-time reduction.";
	mes "Original Beelzebub Card: 30% cast-time reduction.";
	mes "The 20% upgrade consumes one loose Torn Card.";
	mes "The 30% upgrade consumes one loose Partially Restored Card.";
	mes "You can repeat any stage, even using cards acquired through trade.";
	mes "Once a card is inserted into equipment, I cannot extract it.";
	mes "Whether a card can be traded, sold or stored is governed by its item definition.";
	close;

L_Torn:
	mes "[Old Woman]";
	mes "Create ^008800" + getitemname(.TornId) + "^000000 (10%).";
	mes "Required materials:";
	for (.@i = 0; .@i < getarraysize(.TornIds); .@i++)
		mes " - " + .TornQty[.@i] + "x " + getitemname(.TornIds[.@i]);
	mes " - " + .TornZeny + " Zeny";
	next;
	if (select("Create this card:Cancel") != 1) close;

	// Recheck immediately before committing; no more dialogue pauses.
	if (#GR_BerzTxn) close;
	if (Zeny < .TornZeny) {
		mes "[Old Woman]";
		mes "You do not have enough Zeny.";
		close;
	}
	for (.@i = 0; .@i < getarraysize(.TornIds); .@i++) {
		if (countitem(.TornIds[.@i]) < .TornQty[.@i]) {
			mes "[Old Woman]";
			mes "You are missing some of the required materials.";
			close;
		}
	}
	if (!checkweight(.TornId, 1)) {
		mes "[Old Woman]";
		mes "Please free an inventory slot and enough carrying capacity.";
		close;
	}

	.@before = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++)
		if (@inventorylist_id[.@i] == .TornId)
			.@before += @inventorylist_amount[.@i];

	#GR_BerzTxn = 1;
	logmes "[GR_BERZ_V4] torn_begin account=" + getcharid(3) + " char=" + getcharid(0);
	Zeny -= .TornZeny;
	for (.@i = 0; .@i < getarraysize(.TornIds); .@i++)
		delitem .TornIds[.@i], .TornQty[.@i];
	getitem .TornId, 1;

	.@after = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++)
		if (@inventorylist_id[.@i] == .TornId)
			.@after += @inventorylist_amount[.@i];
	if (.@after != .@before + 1) {
		logmes "[GR_BERZ_V4] torn_REVIEW_REQUIRED account=" + getcharid(3) + " char=" + getcharid(0) + " before=" + .@before + " after=" + .@after;
		mes "[Old Woman]";
		mes "The exchange needs GM review. Please do not retry.";
		close;
	}

	logmes "[GR_BERZ_V4] torn_complete account=" + getcharid(3) + " char=" + getcharid(0);
	#GR_BerzTxn = 0;
	callfunc "F_GrassRO_ItemCreationAnnounce", .TornId, "Old Woman";
	mes "[Old Woman]";
	mes "Your Torn Berzebub Card is ready!";
	emotion ET_CONGRATULATION;
	close;

L_Restored:
	mes "[Old Woman]";
	mes "Restore a loose ^008800" + getitemname(.TornId) + "^000000 into ^008800" + getitemname(.RestoredId) + "^000000 (20%).";
	mes "The Torn Card will be consumed.";
	mes "Required materials:";
	mes " - 1x " + getitemname(.TornId);
	for (.@i = 0; .@i < getarraysize(.RestoredIds); .@i++)
		mes " - " + .RestoredQty[.@i] + "x " + getitemname(.RestoredIds[.@i]);
	mes " - " + .RestoredZeny + " Zeny";
	next;
	if (select("Restore this card:Cancel") != 1) close;

	if (#GR_BerzTxn) close;
	if (Zeny < .RestoredZeny) {
		mes "[Old Woman]";
		mes "You do not have enough Zeny.";
		close;
	}
	for (.@i = 0; .@i < getarraysize(.RestoredIds); .@i++) {
		if (countitem(.RestoredIds[.@i]) < .RestoredQty[.@i]) {
			mes "[Old Woman]";
			mes "You are missing some of the required materials.";
			close;
		}
	}
	if (!checkweight(.RestoredId, 1)) {
		mes "[Old Woman]";
		mes "Please free an inventory slot and enough carrying capacity.";
		close;
	}

	.@sourceIndex = -1;
	.@before = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] == .TornId && @inventorylist_equip[.@i] == 0 && .@sourceIndex < 0)
			.@sourceIndex = @inventorylist_idx[.@i];
		if (@inventorylist_id[.@i] == .RestoredId)
			.@before += @inventorylist_amount[.@i];
	}
	if (.@sourceIndex < 0) {
		mes "[Old Woman]";
		mes "You need one loose, uncompounded Torn Card in your inventory.";
		close;
	}

	#GR_BerzTxn = 2;
	logmes "[GR_BERZ_V4] restored_begin account=" + getcharid(3) + " char=" + getcharid(0) + " index=" + .@sourceIndex;
	if (!delitemidx(.@sourceIndex, 1)) {
		logmes "[GR_BERZ_V4] restored_delete_rejected account=" + getcharid(3) + " char=" + getcharid(0);
		#GR_BerzTxn = 0;
		mes "[Old Woman]";
		mes "The Torn Card could not be consumed. Nothing else was taken.";
		close;
	}
	Zeny -= .RestoredZeny;
	for (.@i = 0; .@i < getarraysize(.RestoredIds); .@i++)
		delitem .RestoredIds[.@i], .RestoredQty[.@i];
	getitem .RestoredId, 1;

	.@after = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++)
		if (@inventorylist_id[.@i] == .RestoredId)
			.@after += @inventorylist_amount[.@i];
	if (.@after != .@before + 1) {
		logmes "[GR_BERZ_V4] restored_REVIEW_REQUIRED account=" + getcharid(3) + " char=" + getcharid(0) + " before=" + .@before + " after=" + .@after;
		mes "[Old Woman]";
		mes "The exchange needs GM review. Please do not retry.";
		close;
	}

	logmes "[GR_BERZ_V4] restored_complete account=" + getcharid(3) + " char=" + getcharid(0);
	#GR_BerzTxn = 0;
	callfunc "F_GrassRO_ItemCreationAnnounce", .RestoredId, "Old Woman";
	mes "[Old Woman]";
	mes "Your Partially Restored Berzebub Card is ready!";
	emotion ET_CONGRATULATION;
	close;

L_Original:
	mes "[Old Woman]";
	mes "Restore a loose ^008800" + getitemname(.RestoredId) + "^000000 into ^FF0000" + getitemname(.OriginalId) + "^000000 (30%).";
	mes "The Partially Restored Card will be consumed.";
	mes "Required materials:";
	mes " - 1x " + getitemname(.RestoredId);
	for (.@i = 0; .@i < getarraysize(.OriginalIds); .@i++)
		mes " - " + .OriginalQty[.@i] + "x " + getitemname(.OriginalIds[.@i]);
	mes " - " + .OriginalZeny + " Zeny";
	next;
	if (select("Restore the original card:Cancel") != 1) close;

	if (#GR_BerzTxn) close;
	if (Zeny < .OriginalZeny) {
		mes "[Old Woman]";
		mes "You do not have enough Zeny.";
		close;
	}
	for (.@i = 0; .@i < getarraysize(.OriginalIds); .@i++) {
		if (countitem(.OriginalIds[.@i]) < .OriginalQty[.@i]) {
			mes "[Old Woman]";
			mes "You are missing some of the required materials.";
			close;
		}
	}
	if (!checkweight(.OriginalId, 1)) {
		mes "[Old Woman]";
		mes "Please free an inventory slot and enough carrying capacity.";
		close;
	}

	.@sourceIndex = -1;
	.@before = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] == .RestoredId && @inventorylist_equip[.@i] == 0 && .@sourceIndex < 0)
			.@sourceIndex = @inventorylist_idx[.@i];
		if (@inventorylist_id[.@i] == .OriginalId)
			.@before += @inventorylist_amount[.@i];
	}
	if (.@sourceIndex < 0) {
		mes "[Old Woman]";
		mes "You need one loose, uncompounded Partially Restored Card in your inventory.";
		close;
	}

	#GR_BerzTxn = 3;
	logmes "[GR_BERZ_V4] original_begin account=" + getcharid(3) + " char=" + getcharid(0) + " index=" + .@sourceIndex;
	if (!delitemidx(.@sourceIndex, 1)) {
		logmes "[GR_BERZ_V4] original_delete_rejected account=" + getcharid(3) + " char=" + getcharid(0);
		#GR_BerzTxn = 0;
		mes "[Old Woman]";
		mes "The Partially Restored Card could not be consumed. Nothing else was taken.";
		close;
	}
	Zeny -= .OriginalZeny;
	for (.@i = 0; .@i < getarraysize(.OriginalIds); .@i++)
		delitem .OriginalIds[.@i], .OriginalQty[.@i];
	getitem .OriginalId, 1;

	.@after = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++)
		if (@inventorylist_id[.@i] == .OriginalId)
			.@after += @inventorylist_amount[.@i];
	if (.@after != .@before + 1) {
		logmes "[GR_BERZ_V4] original_REVIEW_REQUIRED account=" + getcharid(3) + " char=" + getcharid(0) + " before=" + .@before + " after=" + .@after;
		mes "[Old Woman]";
		mes "The exchange needs GM review. Please do not retry.";
		close;
	}

	logmes "[GR_BERZ_V4] original_complete account=" + getcharid(3) + " char=" + getcharid(0);
	#GR_BerzTxn = 0;
	callfunc "F_GrassRO_ItemCreationAnnounce", .OriginalId, "Old Woman";
	mes "[Old Woman]";
	mes "Your original Beelzebub Card is ready!";
	emotion ET_CONGRATULATION;
	close;

OnInit:
	.TornId = 39999;
	.RestoredId = 40000;
	.OriginalId = 4145;
	.TornZeny = 1000000;
	.RestoredZeny = 3000000;
	.OriginalZeny = 7000000;

	setarray .TornIds[0], 714,1061,1095,618,1551,727;
	setarray .TornQty[0], 5,100,150,5,1,10;

	setarray .RestoredIds[0], 714,1061,1095,618,1551,1614,727;
	setarray .RestoredQty[0], 15,250,350,10,1,1,20;

	setarray .OriginalIds[0], 714,1061,1095,618,1551,1614,727;
	setarray .OriginalQty[0], 40,650,850,25,2,2,60;
	end;
}
