// GrassRO - Shared MVP Kill Announcement
// Preserves original message format and World Feed marker.
// Natural spawns pass their mob ID directly.

function	script	F_GR_MVP_Announce	{

	.@mob_id = getarg(0, 0);

	if (.@mob_id <= 0 || getmonsterinfo(.@mob_id, MOB_MVPEXP) <= 0)
		return;

	.@player$ = strcharinfo(0);
	.@party$ = strcharinfo(1);
	.@map$ = strcharinfo(3);
	.@monster$ = getmonsterinfo(.@mob_id, MOB_NAME);

	if (.@party$ == "")
		.@message$ = "[MVP Kill] " + .@player$ + " defeated " + .@monster$ + " at " + .@map$ + "!";
	else
		.@message$ = "[MVP Kill] " + .@player$ + " of party [" + .@party$ + "] defeated " + .@monster$ + " at " + .@map$ + "!";

	announce .@message$, bc_all, 0xFFD700;
	logmes "[GR_WORLD_FEED_MVP_KILL] " + .@message$;

	return;
}

-	script	GR_MVP_Kill_Announcement	-1,{

OnNPCKillEvent:
	callfunc "F_GR_MVP_Announce", killedrid;
	end;
}
