		.@menu$ += ":^00AA00Group 99 Administration^000000";
	.@action = select(.@menu$);

		close;

	clear;

		mes "[Job Master]";
		mes "^FF0000Job Master is locked for this character pending group 99 review.^000000";

		mes "[Job Master]";

	if (SkillPoint > 0) {
		mes "[Job Master]";
		mes "Spend all unused skill points before changing jobs.";

	.@n = 0;

			setarray .@o[0],Job_Swordman,Job_Mage,Job_Archer,Job_Acolyte,Job_Merchant,Job_Thief,Job_Super_Novice,Job_Taekwon,Job_Gunslinger,Job_Ninja;

			// Players reach Baby Novice ONLY via the real adoption system.
			setarray .@o[0],Job_Baby_Swordman,Job_Baby_Mage,Job_Baby_Archer,Job_Baby_Acolyte,Job_Baby_Merchant,Job_Baby_Thief,Job_Super_Baby,Job_Baby_Taekwon,Job_Baby_Gunslinger,Job_Baby_Ninja;
			.@n = 10;
			break;
		case Job_Novice_High:
			// Official Valkyrie stores ADVJOB; old custom Job Master stored lastJob.
			.@lineage = ADVJOB;
			if (.@lineage == 0 && lastJob > 0)
				.@lineage = lastJob + Job_Novice_High;
			if (lastJob > 0 && ADVJOB > 0 && (lastJob + Job_Novice_High) != ADVJOB)
				.@lineage = 0;
			if (.@lineage > 0 && eaclass(.@lineage) != -1 && (eaclass(.@lineage) & EAJL_2) && (eaclass(.@lineage) & EAJL_UPPER)) {
				.@to = roclass((eaclass(.@lineage) & EAJ_BASEMASK) | EAJL_UPPER);

				if (valkyrie_Q == 0 && ADVJOB == 0) {
					.@o[.@n] = Job_Novice_High;

					mes "[Job Master]";

			// Standard, High and Baby first-class families, including Taekwon.
			switch (Class) {
				case Job_Swordman: case Job_Mage: case Job_Archer:
				case Job_Acolyte: case Job_Merchant: case Job_Thief:
				case Job_Swordman_High: case Job_Mage_High: case Job_Archer_High:

		.@discount = (.@base * $JM_Discount) / 100;

		.@with = (.@pack == 2);

		mes "There is no eligible, priced equipment reward for this class. Job change only.";

		mes "Balance: " + callfunc("F_InsertComma",Zeny) + "z";

		close;

	JM_TxTarget = .@target;
	JM_TxCharge = .@total;
	JM_TxItem = (.@with ? .@item : 0);

		// DO NOT modify npc/jobs/valkyrie.txt itself.
		callfunc "F_ClearJobVar";
		lastJob = .@source;

		if (ADVJOB == Job_Lord_Knight2) ADVJOB = Job_Lord_Knight;
		if (ADVJOB == Job_Paladin2) ADVJOB = Job_Paladin;

		logmes "JM ERROR jobchange failed char="+getcharid(0)+" expected="+.@target+" actual="+Class;
		mes "[Job Master]";
		mes "^FF0000Unexpected result. Contact group 99. Do not retry.^000000";
		close;
	}

		resetlvl(1);
		MISC_QUEST = MISC_QUEST & ~1024;
		skill "NV_FIRSTAID",1,SKILL_PERM;
		skill "NV_TRICKDEAD",1,SKILL_PERM;

	} else {
		// Sync old Job Master lineage after an official Valkyrie rebirth.
		if ((eaclass(.@target)&EAJL_UPPER) && ADVJOB > 0 && lastJob == 0)
			lastJob = ADVJOB - Job_Novice_High;

			callfunc "F_GetPlatinumSkills";
	}
	if (.@with && .@item > 0) {
		getitem .@item,1;
		logmes "JM REWARD char="+getcharid(0)+" item="+.@item+" amount=1 (may drop if overweight)";
	}
	JM_LastTarget = .@target;
	JM_LastPaid = .@total;
	JM_LastWhen = gettimetick(2);
	logmes "JM COMPLETE char="+getcharid(0)+" src="+.@source+" dst="+Class+" zeny="+Zeny;
	JM_TxState = 0;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	mes "[Job Master]";
	mes "^00AA00Advancement completed!^000000 You are now " + jobname(Class) + ".";
	mes "Paid: " + .@total + "z. Remaining: " + Zeny + "z.";
	mes "No teleport was performed.";
	close;

// ----- Class-based pricing: all non-listed classes are rejected. --------
function BaseFee {
	.@j = getarg(0);
	if (.@j == Job_Novice_High) return 1285000;
	if (.@j == Job_Super_Novice || .@j == Job_Super_Baby || .@j == Job_Ninja || .@j == Job_Baby_Ninja)
		return 10000;
	if (.@j == Job_Taekwon || .@j == Job_Baby_Taekwon) return 1000;
	if (.@j == Job_Gunslinger || .@j == Job_Baby_Gunslinger) return 5000;
	if (.@j == Job_Swordman_High || .@j == Job_Mage_High || .@j == Job_Archer_High ||
		.@j == Job_Acolyte_High || .@j == Job_Merchant_High || .@j == Job_Thief_High)
		return 10000;
	if (.@j == Job_Swordman || .@j == Job_Mage || .@j == Job_Archer ||
		.@j == Job_Acolyte || .@j == Job_Merchant || .@j == Job_Thief ||
		.@j == Job_Baby_Swordman || .@j == Job_Baby_Mage || .@j == Job_Baby_Archer ||
		.@j == Job_Baby_Acolyte || .@j == Job_Baby_Merchant || .@j == Job_Baby_Thief)
		return 5000;
	.@e = eaclass(.@j);
	if (.@e == -1 || (.@e & (EAJL_THIRD|EAJL_FOURTH)) || !(.@e & EAJL_2)) return -1;
	if (.@e & EAJL_UPPER) return 30000;
	.@base = roclass(.@e & EAJ_UPPERMASK);
	switch (.@base) {
		case Job_Knight: return 50000;
		case Job_Crusader: return 75000;
		case Job_Priest: return 40000;
		case Job_Monk: return 100000;
		case Job_Wizard: return 80000;
		case Job_Sage: return 150000;
		case Job_Hunter: return 50000;
		case Job_Bard: return 90000;
		case Job_Dancer: return 60000;
		case Job_Blacksmith: return 70000;
		case Job_Alchemist: return 150000;
		case Job_Assassin: return 60000;
		case Job_Rogue: return 90000;
		case Job_Star_Gladiator: return 125000;
		case Job_Soul_Linker: return 100000;
	}
	return -1;
}
function EffectiveFee {
	.@j = getarg(0);
	.@default = BaseFee(.@j);
	if (.@default < 0) return -1;
	if (getd("$JM_FeeOverrideOn_" + .@j))
		return getd("$JM_FeeOverride_" + .@j);
	return .@default;
}
// ----- Item mapping mirrors upstream classic Job Master rewards. --------
function RewardItem {
	.@j = getarg(0);
	if (.@j == Job_Novice_High) return 0;
	if (.@j == Job_Super_Novice || .@j == Job_Super_Baby) return 1208;
	.@e = eaclass(.@j);
	if (.@e == -1 || (.@e & (EAJL_THIRD|EAJL_FOURTH))) return 0;
	if (.@e & EAJL_2) {
		switch (roclass(.@e & EAJ_UPPERMASK)) {
			case Job_Knight: return 1163;
			case Job_Priest: return 1522;
			case Job_Wizard: return 1617;
			case Job_Blacksmith: return 1360;
			case Job_Hunter: return 1718;
			case Job_Assassin: return 1254;
			case Job_Crusader: return 1410;
			case Job_Monk: return 1807;
			case Job_Sage: return 1550;
			case Job_Rogue: return 1222;
			case Job_Alchemist: return 1126;
			case Job_Bard: return 1907;
			case Job_Dancer: return 1960;
			case Job_Star_Gladiator: return 1550;
			case Job_Soul_Linker: return 1617;
		}
	}
	switch (roclass(.@e & EAJ_BASEMASK)) {
		case Job_Swordman: return 1108;
		case Job_Mage: return 1602;
		case Job_Archer: return 1705;
		case Job_Acolyte: return 1505;
		case Job_Merchant: return 1302;
		case Job_Thief: return 1208;
		case Job_Gunslinger: return 13101;
		case Job_Ninja: return 13010;
	}
	return 0;
}
function IsMappedReward {
	.@id = getarg(0);
	for (.@k=0; .@k < getarraysize(.RewardWhitelist); .@k++)
		if (.RewardWhitelist[.@k] == .@id) return 1;
	return 0;
}
function ItemCharge {
	.@id = getarg(0);
	if (.@id <= 0 || !IsMappedReward(.@id) || getiteminfo(.@id,ITEMINFO_ID) != .@id) return -1;
	// An explicit GM override is allowed to equal 0 (free equipment).
	if (getd("$JM_EqOverrideOn_" + .@id))
		return getd("$JM_EqOverride_" + .@id);
	.@buy = getiteminfo(.@id,ITEMINFO_BUY);
	if (.@buy <= 0) return -1;
	.@price = .@buy / 2; // integer division = round down
	if (.@price < 1) .@price = 1; // only GM may make equipment free
	return .@price;
}
// ----- Final validation is also called BEFORE payment. ------------------
function ValidateChange {
	.@from = getarg(0);
	.@to = getarg(1);
	if (Class != .@from || BaseFee(.@to) < 0) return 0;
	if (eaclass(.@to) == -1 || (eaclass(.@to)&(EAJL_THIRD|EAJL_FOURTH))) return 0;
	// A Baby class must have a real family record, not just a Baby class ID.
	if ((eaclass(.@to)&EAJL_BABY) && (!getmotherid() || !getfatherid())) return 0;
	if (JM_TxState != 0) return 0;
	if (.@to == Job_Novice_High) {
		if (!(Class >= Job_Knight && Class <= Job_Crusader2) || (eaclass()&(EAJL_BABY|EAJL_UPPER))) return 0;
		if (Upper != 0 || ADVJOB != 0 || valkyrie_Q != 0) return 0;
		return BaseLevel >= 99 && JobLevel >= 50;
	}
	if (.@from == Job_Novice || .@from == Job_Baby || .@from == Job_Novice_High) {
		if (JobLevel < 10) return 0;
		if ((.@to == Job_Super_Novice || .@to == Job_Super_Baby) && BaseLevel < 45) return 0;
		if (.@from == Job_Novice_High) {
			.@lineage = ADVJOB;
			if (.@lineage == 0 && lastJob > 0) .@lineage = lastJob + Job_Novice_High;
			if (.@lineage <= 0 || eaclass(.@lineage) == -1 || !(eaclass(.@lineage)&EAJL_UPPER) ||
				!(eaclass(.@lineage)&EAJL_2) ||
				(lastJob > 0 && ADVJOB > 0 && lastJob+Job_Novice_High != ADVJOB)) return 0;
			return .@to == roclass((eaclass(.@lineage)&EAJ_BASEMASK)|EAJL_UPPER);
		}
		if (.@from == Job_Novice) {
			if (eaclass(.@to)&(EAJL_BABY|EAJL_UPPER|EAJL_2))
				return .@to == Job_Super_Novice;
			return .@to == Job_Super_Novice || .@to == Job_Taekwon ||
				.@to == Job_Gunslinger || .@to == Job_Ninja ||
				(.@to >= Job_Swordman && .@to <= Job_Thief);
		}
		if (.@from == Job_Baby) {
			return .@to == Job_Super_Baby || .@to == Job_Baby_Taekwon ||
				.@to == Job_Baby_Gunslinger || .@to == Job_Baby_Ninja ||
				.@to == Job_Baby_Swordman || .@to == Job_Baby_Mage ||
				.@to == Job_Baby_Archer || .@to == Job_Baby_Acolyte ||
				.@to == Job_Baby_Merchant || .@to == Job_Baby_Thief;
		}
	}
	if (JobLevel < 40) return 0;
	.@src = eaclass(.@from);
	.@dst = eaclass(.@to);
	if (.@src == -1 || .@dst == -1) return 0;
	if ((.@src&EAJ_BASEMASK) != (.@dst&EAJ_BASEMASK)) return 0;
	if ((.@src&EAJL_BABY) != (.@dst&EAJL_BABY)) return 0;
	if ((.@src&EAJL_UPPER) != (.@dst&EAJL_UPPER)) return 0;
	if (!(.@dst&EAJL_2) || (.@src&EAJL_2)) return 0;
	if (.@src&EAJL_UPPER) {
		.@lineage = ADVJOB;
		if (.@lineage == 0 && lastJob > 0) .@lineage = lastJob + Job_Novice_High;
		if (.@lineage <= 0 || .@lineage != .@to ||
			(lastJob > 0 && ADVJOB > 0 && lastJob + Job_Novice_High != ADVJOB)) return 0;
	}
	return 1;
}
// ----- Player-facing fee guide: short pages, categorized by topic. -------

			return;
		clear;
		switch (.@section) {
			case 1:
				mes "[Job Master]";
				mes "^3366CCSTANDARD ADVANCEMENT^000000";
				mes "First Job: ^0055FF" + callfunc("F_InsertComma",BaseFee(Job_Swordman)) + "z^000000";
				mes "High First Job: ^0055FF" + callfunc("F_InsertComma",BaseFee(Job_Swordman_High)) + "z^000000";
				mes "Transcendent Second: ^0055FF" + callfunc("F_InsertComma",BaseFee(Job_Lord_Knight)) + "z^000000";
				next;
				mes "[Job Master]";
				mes "^3366CCREBIRTH^000000";
				mes "Current fee: ^0055FF" + callfunc("F_InsertComma",EffectiveFee(Job_Novice_High)) + "z^000000";
				mes "Requires ^FF8800Base 99 / Job 50^000000.";
				mes "^777777GM overrides may change individual class fees.^000000";
				next;
				break;

// ----- Admin: only exact group ID 99. Settings survive reloads. ---------
function AdminPanel {
	if (getgroupid() != 99) return;
	while (1) {
		mes "[Job Master: Group 99]";
		mes "All changes are global and apply to future purchases.";
		.@action = select("View/Change Job Fee:Change Rebirth Fee:Equipment Price Override:Max Job Discount:Toggle Platinum Skills:Review/Unlock Character:Reset Overrides:Exit");
		if (getgroupid() != 99) close;
		if (.@action == 8) return;
		if (.@action == 6) {
			HandleReview();
			return;
		}
		if (.@action == 4) {
			mes "[Job Master]";
			mes "Current discount: " + $JM_Discount + "%. Set 0-50%.";
			input .@new;
			if (.@new < 0 || .@new > 50) { mes "Invalid range."; close; }
			if (select("Confirm change:Cancel") == 1 && getgroupid() == 99) {
				logmes "JM ADMIN discount GM="+getcharid(0)+" old="+$JM_Discount+" new="+.@new;
				$JM_Discount = .@new;
			}
			close;
		}
		if (.@action == 5) {
			mes "[Job Master]";
			mes "Automatic Platinum Skills: " + ($JM_Platinum ? "Enabled" : "Disabled");
			if (select("Toggle globally:Cancel") == 1 && getgroupid() == 99) {
				$JM_Platinum = !$JM_Platinum;
				logmes "JM ADMIN platinum GM="+getcharid(0)+" value="+$JM_Platinum;
			}
			close;
		}
		if (.@action == 1 || .@action == 2) {
			mes "[Job Master]";
			if (.@action == 2) .@id = Job_Novice_High;
			else {
				mes "Enter a permitted destination job ID.";
				input .@id;
			}
			if (BaseFee(.@id) < 0) { mes "Invalid or prohibited destination."; close; }
			mes "Class: " + jobname(.@id);
			mes "Default fee: " + BaseFee(.@id) + "z; active fee: " + EffectiveFee(.@id) + "z.";
			.@choice = select("Set final fee:Reset class fee to default:Cancel");
			if (.@choice == 3) close;
			if (getgroupid() != 99) close;
			if (.@choice == 2) {
				setd "$JM_FeeOverrideOn_"+.@id,0;
				logmes "JM ADMIN reset job fee GM="+getcharid(0)+" id="+.@id;
				close;
			}
			mes "Set new final job fee (1 to 10,000,000z):";
			input .@new;
			if (.@new < 1 || .@new > 10000000) { mes "Invalid fee."; close; }
			mes "Change " + jobname(.@id) + " to " + .@new + "z?";
			if (select("Confirm:Cancel") == 1 && getgroupid() == 99) {
				logmes "JM ADMIN job fee GM="+getcharid(0)+" id="+.@id+" old="+EffectiveFee(.@id)+" new="+.@new;
				setd "$JM_FeeOverride_"+.@id,.@new;
				setd "$JM_FeeOverrideOn_"+.@id,1;
			}
			close;
		}
		if (.@action == 3) {
			mes "[Job Master]";
			mes "Enter an equipment reward item ID from the Job Master mapping.";
			input .@id;
			if (!IsMappedReward(.@id) || getiteminfo(.@id,ITEMINFO_ID) != .@id) {
				mes "Invalid/unmapped item. Override refused.";
				close;
			}
			mes "Item: " + getitemname(.@id);
			mes "Effective equipment price: " + ItemCharge(.@id) + "z (-1 means unavailable).";
			.@choice = select("Set exact final equipment price:Remove override:Cancel");
			if (.@choice == 3) close;
			if (.@choice == 2 && getgroupid() == 99) {
				setd "$JM_EqOverrideOn_"+.@id,0;
				logmes "JM ADMIN reset item price GM="+getcharid(0)+" item="+.@id;
				close;
			}
			mes "Enter 0 to 100,000,000z. Zero makes this reward FREE.";
			input .@new;
			if (.@new < 0 || .@new > 100000000) { mes "Invalid price."; close; }
			if (.@new == 0)
				mes "^FF0000WARNING: This will make " + getitemname(.@id) + " FREE during eligible advancement.^000000";
			mes "Set FINAL item charge to " + .@new + "z?";
			if (select("Confirm override:Cancel") == 1 && getgroupid() == 99) {
				logmes "JM ADMIN equipment price GM="+getcharid(0)+" item="+.@id+" old="+ItemCharge(.@id)+" new="+.@new;
				setd "$JM_EqOverride_"+.@id,.@new;
				setd "$JM_EqOverrideOn_"+.@id,1;
			}
			close;
		}
		if (.@action == 7) {
			mes "[Job Master]";
			mes "Reset ALL price overrides and restore 20% max discount and Platinum Skills enabled?";
			mes "^FF0000This also removes intentional free-equipment overrides.^000000";
			if (select("Continue to final confirmation:Cancel") != 1) close;
			if (select("CONFIRM RESET ALL:Cancel") != 1 || getgroupid() != 99) close;
			for (.@i=0; .@i<getarraysize(.AllowedJobs); .@i++)
				setd "$JM_FeeOverrideOn_"+.AllowedJobs[.@i],0;
			for (.@i=0; .@i<getarraysize(.RewardWhitelist); .@i++)
				setd "$JM_EqOverrideOn_"+.RewardWhitelist[.@i],0;
			$JM_Discount = 20;
			$JM_Platinum = 1;
			logmes "JM ADMIN RESET ALL GM="+getcharid(0);
			close;
		}
	}
	return;
}
// ----- Review ONLINE character, no automatic refund or item spawn. -----
function HandleReview {
	if (getgroupid() != 99) return;
	mes "[Job Master: Review]";
	mes "Character must be ONLINE; enter their exact character name.";
	input .@who$;
	.@target_aid = getcharid(3,.@who$);
	if (.@target_aid <= 0) { mes "Character is not online or not found."; close; }
	.@gm_aid = getcharid(3);
	.@gm_cid = getcharid(0);
	mes "After viewing the reported transaction, investigate independently.";
	mes "^FF0000Only resolve after verifying and documenting the outcome. This panel cannot refund or spawn items.^000000";
	.@decision = select("Inspect only:Resolve and unlock:Cancel");
	if (.@decision == 3) close;
	if (.@decision == 2) {
		mes "Enter your investigation and compensation note:";
		input .@note$;
		if (.@note$ == "") { mes "Resolution note required."; close; }
		mes "Unlock this character based on your investigation?";
		if (select("Final confirmation:Cancel") != 1) close;
	}
	if (getgroupid() != 99) close;
	close2;
	// RID switch occurs ONLY after the original GM dialogue is closed.
	if (!attachrid(.@target_aid)) {
		attachrid .@gm_aid;
		dispbottom "JM: Target went offline. No changes made.";
		end;
	}
	if (strcharinfo(0) != .@who$) {
		detachrid;
		attachrid .@gm_aid;
		dispbottom "JM: Character identity mismatch. No changes made.";
		end;
	}
	.@state = JM_TxState;
	.@source = JM_TxSource;
	.@dest = JM_TxTarget;
	.@paid = JM_TxCharge;
	.@item = JM_TxItem;
	.@before = JM_TxBefore;
	.@when = JM_TxWhen;
	.@actual = Class;
	.@zeny = Zeny;
	if (.@decision == 2 && .@state == 1) {
		logmes "JM ADMIN RESOLVE gm_cid="+.@gm_cid+" target_cid="+getcharid(0)+" orig="+.@source+" dest="+.@dest+" actual="+Class+" paid="+.@paid+" note="+.@note$;
		JM_LastResolution$ = .@note$;
		JM_LastReviewer = .@gm_cid;
		JM_TxState = 0;
	}
	detachrid;
	if (!attachrid(.@gm_aid)) end;
	dispbottom "JM REVIEW "+.@who$+": status="+.@state+" source="+.@source+" dest="+.@dest+" actual="+.@actual;
	dispbottom "JM REVIEW: quoted="+.@paid+"z initial="+.@before+"z current="+.@zeny+"z item="+.@item+" time="+.@when;
	if (.@decision == 2 && .@state == 1)
		dispbottom "JM: Resolution recorded and player Job Master unlocked.";
	else if (.@decision == 2)
		dispbottom "JM: Character was not locked; no unlock performed.";
	end;
}
