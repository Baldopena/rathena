//============================================================
//= Grass Ro | Wakim | Phase 3 — GM-only protected Sell All transaction candidate
//= Native Buy / Sell + Identify All (quantity-based billing)
//= Sell All GM-only: gated actual deletion/payout; deployment requires safety validation.
//= Requires Hunting Arrow Item ID 1774 in the effective item DB.
//= NEVER register this alongside another Wakim NPC at the same map position.
//============================================================

prontera,146,172,4	script	Wakim#GrassRo	582,{
	if (getgmlevel() < 99) {
		mes "[Wakim]";
		mes "This service is currently under GM testing.";
		close;
	}
	mes "[Wakim]";
	mes "Welcome to Wakim's services.";
	if (getgmlevel() >= 99) {
		.@choice = select("Buy Items:Sell Items:Sell All Inventory:Identify All Items:Administration:Cancel");
	} else {
		.@choice = select("Buy Items:Sell Items:Sell All Inventory:Identify All Items:Cancel");
	}

	switch (.@choice) {
	case 1:
		close2;
		callshop "Wakim_Stock#GrassRo",1;
		end;
	case 2:
		close2;
		callshop "Wakim_Stock#GrassRo",2;
		end;
	case 3:
		goto L_SellAll;
	case 4:
		goto L_Identify;
	case 5:
		if (getgmlevel() < 99)
			close;
		goto L_Admin;
	default:
		close;
	}

L_SellAll:
	// GM-only operational test. No exposure to ordinary players in this version.
	// Native NoSell list MUST be generated against the effective PRE-RE item DB.
	if (getgmlevel() < 99) close;
	if (!$GR_WakimSaleEnabled || !.NoSellReady || !$GR_WakimExcludeReady) {
		mes "[Wakim]";
		mes "^FF0000Protected Sell All is not enabled.^000000";
		mes "Generate and verify native NoSell registry, then enable GM testing.";
		close;
	}
	if (WakimSellIncident) {
		mes "[Wakim]";
		mes "^FF0000Unresolved Sell All incident. Contact an administrator.^000000";
		close;
	}
	if (@WakimOperationBusy || $GR_WakimEditBusy) {
		mes "[Wakim]";
		mes "Another operation is running; retry.";
		close;
	}
	// Ensure compact exclusion array and integrity checksum are self-consistent.
	if ($GR_WakimExcludeCount < 0 || $GR_WakimExcludeCount > 40) {
		mes "[Wakim]"; mes "Exclusion registry invalid; Sell All suspended."; close;
	}
	.@sum = 1;
	for (.@j=0; .@j < $GR_WakimExcludeCount; ++.@j) {
		if ($GR_WakimExcludes[.@j] <= 0) {
			mes "[Wakim]"; mes "Exclusion registry contains invalid entries."; close;
		}
		for (.@k=0; .@k < .@j; ++.@k)
			if ($GR_WakimExcludes[.@k] == $GR_WakimExcludes[.@j]) {
				mes "[Wakim]"; mes "Duplicate exclusion ID; operation suspended."; close;
			}
		.@sum += $GR_WakimExcludes[.@j] * (.@j + 1);
	}
	if (.@sum != $GR_WakimExcludeSum) {
		mes "[Wakim]"; mes "Exclusion registry checksum mismatch; suspended."; close;
	}
	.@revision = $GR_WakimExcludeRev;
	.@oldZeny = Zeny;
	.@oldOC = getskilllv("MC_OVERCHARGE");
	.@rate = (.@oldOC > 0) ? 5 + .@oldOC * 2 - (.@oldOC == 10 ? 1 : 0) : 0;
	getinventorylist;
	.@count = @inventorylist_count;
	.@subtotal = 0;
	.@n = 0;
	.@special = 0;
	.@skip = 0;
	.@units = 0;
	// Snapshot ALL observable inventory attributes, even for excluded entries.
	// An item unique_id and Gear Switch state are not script-visible.
	for (.@i=0; .@i < .@count; ++.@i) {
		.@sIdx[.@i] = @inventorylist_idx[.@i];
		.@sId[.@i] = @inventorylist_id[.@i];
		.@sAmt[.@i] = @inventorylist_amount[.@i];
		.@sEquip[.@i] = @inventorylist_equip[.@i];
		.@sFav[.@i] = @inventorylist_favorite[.@i];
		.@sIdent[.@i] = @inventorylist_identify[.@i];
		.@sBound[.@i] = @inventorylist_bound[.@i];
		.@sExpire[.@i] = @inventorylist_expire[.@i];
		.@sRefine[.@i] = @inventorylist_refine[.@i];
		.@sAttr[.@i] = @inventorylist_attribute[.@i];
		.@sEnchant[.@i] = @inventorylist_enchantgrade[.@i];
		.@sCard1[.@i] = @inventorylist_card1[.@i];
		.@sCard2[.@i] = @inventorylist_card2[.@i];
		.@sCard3[.@i] = @inventorylist_card3[.@i];
		.@sCard4[.@i] = @inventorylist_card4[.@i];
		.@sTradable[.@i] = @inventorylist_tradable[.@i];
		.@sOptId1[.@i] = @inventorylist_option_id1[.@i];
		.@sOptVal1[.@i] = @inventorylist_option_value1[.@i];
		.@sOptParam1[.@i] = @inventorylist_option_parameter1[.@i];
		.@sOptId2[.@i] = @inventorylist_option_id2[.@i];
		.@sOptVal2[.@i] = @inventorylist_option_value2[.@i];
		.@sOptParam2[.@i] = @inventorylist_option_parameter2[.@i];
		.@sOptId3[.@i] = @inventorylist_option_id3[.@i];
		.@sOptVal3[.@i] = @inventorylist_option_value3[.@i];
		.@sOptParam3[.@i] = @inventorylist_option_parameter3[.@i];
		.@sOptId4[.@i] = @inventorylist_option_id4[.@i];
		.@sOptVal4[.@i] = @inventorylist_option_value4[.@i];
		.@sOptParam4[.@i] = @inventorylist_option_parameter4[.@i];
		.@sOptId5[.@i] = @inventorylist_option_id5[.@i];
		.@sOptVal5[.@i] = @inventorylist_option_value5[.@i];
		.@sOptParam5[.@i] = @inventorylist_option_parameter5[.@i];
		.@eligible[.@i] = 0;
		if (.@sId[.@i] <= 0 || .@sAmt[.@i] <= 0 || .@sEquip[.@i] || .@sFav[.@i] || .@sIdent[.@i] != 1 || .@sBound[.@i] || .@sExpire[.@i]) {
			.@skip++; continue;
		}
		if (getiteminfo(.@sId[.@i], ITEMINFO_TYPE) == IT_PETEGG) { .@skip++; continue; }
		if (inarray(.NoSell, .@sId[.@i]) >= 0) { .@skip++; continue; }
		.@blocked = 0;
		for (.@j=0; .@j < $GR_WakimExcludeCount; ++.@j)
			if ($GR_WakimExcludes[.@j] == .@sId[.@i]) .@blocked = 1;
		// Cards inserted into equipment can carry their own NoSell restriction.
		if (inarray(.NoSell, .@sCard1[.@i]) >= 0 || inarray(.NoSell, .@sCard2[.@i]) >= 0 || inarray(.NoSell, .@sCard3[.@i]) >= 0 || inarray(.NoSell, .@sCard4[.@i]) >= 0) .@blocked = 1;
		if (.@blocked) { .@skip++; continue; }
		.@raw = getiteminfo(.@sId[.@i], ITEMINFO_SELL);
		if (.@raw < 0) { .@skip++; continue; }
		.@unit = (.@raw * (100 + .@rate)) / 100;
		if (.@unit < 0) .@unit = 0; // Production min_shop_sell = 0.
		.@line = .@unit * .@sAmt[.@i];
		if (.@line < 0 || .@subtotal > 50000000 || .@line > 50000000 - .@subtotal) {
			mes "[Wakim]"; mes "^FF0000Pricing overflow or maximum payout exceeded.^000000"; close;
		}
		.@eligible[.@i] = 1;
		.@sUnit[.@i] = .@unit;
		.@sLine[.@i] = .@line;
		.@subtotal += .@line;
		.@n++;
		.@units += .@sAmt[.@i];
		.@type = getiteminfo(.@sId[.@i], ITEMINFO_TYPE);
		if (.@unit == 0 || .@type == IT_WEAPON || .@type == IT_ARMOR || .@type == IT_CARD || .@sRefine[.@i] || .@sEnchant[.@i] || .@sCard1[.@i] || .@sCard2[.@i] || .@sCard3[.@i] || .@sCard4[.@i] || .@sOptId1[.@i] || .@sOptId2[.@i] || .@sOptId3[.@i] || .@sOptId4[.@i] || .@sOptId5[.@i]) .@special = 1;
	}
	if (.@n == 0) { mes "[Wakim]"; mes "No eligible inventory items."; close; }
	// Receipt must fit a usable NPC dialog: reject rather than truncate.
	if (.@n > 100) { mes "[Wakim]"; mes "Too many lines for a safe receipt (max 100); sale canceled."; close; }
	.@bonus = (.@subtotal * 10) / 100;
	.@total = .@subtotal + .@bonus;
	if (.@total > 50000000 || .@total < 0 || .@oldZeny > 1000000000 - .@total) {
		mes "[Wakim]"; mes "^FF0000Payout or character Zeny cap exceeded.^000000"; close;
	}
	mes "[Wakim - Sell All Receipt]";
	mes "^FF0000TEST TRANSACTION: ITEM DELETION IS REAL.^000000";
	mes "^FF0000Favorite any equipment you want to keep.^000000";
	mes "^FF0000Unequipped Gear Switch presets are NOT protected.^000000";
	mes "--------------------------------";
	for (.@i=0; .@i < .@count; ++.@i) {
		if (!.@eligible[.@i]) continue;
		mes getitemname(.@sId[.@i]) + " [" + .@sId[.@i] + "] x" + .@sAmt[.@i];
		mes "  " + .@sUnit[.@i] + "z x" + .@sAmt[.@i] + " = " + .@sLine[.@i] + "z";
	}
	mes "--------------------------------";
	mes "Items: " + .@units + " units / " + .@n + " slots. Excluded: " + .@skip;
	mes "Overcharge: " + .@rate + "% (per unit)";
	mes "Resale: " + .@subtotal + "z";
	mes "Extra 10%: " + .@bonus + "z";
	mes "^00AA00Total: " + .@total + "z^000000";
	if (.@special) mes "^FF0000Includes gear, cards, modified items, or zero-price items.^000000";
	next;
	if (select("Proceed to final confirmation:Cancel") != 1) close;
	mes "[Wakim - Final Confirmation]";
	mes "^FF0000THIS PERMANENTLY DELETES ALL LISTED ITEMS.^000000";
	mes "There is no automatic rollback on incomplete sales.";
	mes "Total payout: " + .@total + "z";
	if (.@special) mes "^FF0000Double-check your equipment and cards.^000000";
	if (select("CONFIRM DELETE AND SELL:Cancel") != 1) close;
	// No yield commands past this point until completion.
	if (@WakimOperationBusy || $GR_WakimEditBusy || !$GR_WakimSaleEnabled || !.NoSellReady || !$GR_WakimExcludeReady || $GR_WakimExcludeRev != .@revision || getgmlevel() < 99 || getskilllv("MC_OVERCHARGE") != .@oldOC || Zeny != .@oldZeny) {
		mes "[Wakim]"; mes "Transaction parameters changed; canceled."; close;
	}
	@WakimOperationBusy = 1;
	getinventorylist;
	.@changed = (@inventorylist_count != .@count);
	if (!.@changed) {
		for (.@i=0; .@i < .@count; ++.@i) {
			if (@inventorylist_idx[.@i] != .@sIdx[.@i] || @inventorylist_id[.@i] != .@sId[.@i] || @inventorylist_amount[.@i] != .@sAmt[.@i] || @inventorylist_equip[.@i] != .@sEquip[.@i] || @inventorylist_favorite[.@i] != .@sFav[.@i] || @inventorylist_identify[.@i] != .@sIdent[.@i] || @inventorylist_bound[.@i] != .@sBound[.@i] || @inventorylist_expire[.@i] != .@sExpire[.@i] || @inventorylist_refine[.@i] != .@sRefine[.@i] || @inventorylist_attribute[.@i] != .@sAttr[.@i] || @inventorylist_enchantgrade[.@i] != .@sEnchant[.@i] || @inventorylist_card1[.@i] != .@sCard1[.@i] || @inventorylist_card2[.@i] != .@sCard2[.@i] || @inventorylist_card3[.@i] != .@sCard3[.@i] || @inventorylist_card4[.@i] != .@sCard4[.@i] || @inventorylist_tradable[.@i] != .@sTradable[.@i] || @inventorylist_option_id1[.@i] != .@sOptId1[.@i] || @inventorylist_option_value1[.@i] != .@sOptVal1[.@i] || @inventorylist_option_parameter1[.@i] != .@sOptParam1[.@i] || @inventorylist_option_id2[.@i] != .@sOptId2[.@i] || @inventorylist_option_value2[.@i] != .@sOptVal2[.@i] || @inventorylist_option_parameter2[.@i] != .@sOptParam2[.@i] || @inventorylist_option_id3[.@i] != .@sOptId3[.@i] || @inventorylist_option_value3[.@i] != .@sOptVal3[.@i] || @inventorylist_option_parameter3[.@i] != .@sOptParam3[.@i] || @inventorylist_option_id4[.@i] != .@sOptId4[.@i] || @inventorylist_option_value4[.@i] != .@sOptVal4[.@i] || @inventorylist_option_parameter4[.@i] != .@sOptParam4[.@i] || @inventorylist_option_id5[.@i] != .@sOptId5[.@i] || @inventorylist_option_value5[.@i] != .@sOptVal5[.@i] || @inventorylist_option_parameter5[.@i] != .@sOptParam5[.@i]) {
				.@changed = 1; break;
			}
		}
	}
	if (.@changed) {
		@WakimOperationBusy = 0;
		mes "[Wakim]"; mes "Inventory changed; no items sold."; close;
	}
	// PERSISTENT PRE-DELETE INCIDENT JOURNAL. IMPORTANT: char-reg saving is not
	// transactionally durable; a map-server crash before persistence may lose it.
	WakimSellIncident = 1;
	WakimSellEpoch = gettimetick(2);
	WakimSellExpected = .@n;
	WakimSellDeleted = 0;
	logmes "WAKIM SELL START cid=" + getcharid(0) + " slots=" + .@n + " total=" + .@total + " rev=" + .@revision;
	for (.@i=0; .@i < .@count; ++.@i) {
		if (!.@eligible[.@i]) continue;
		// Index-targeted deletion is essential to protect equipped duplicate IDs.
		logmes "WAKIM SELL DELETE_ATTEMPT cid=" + getcharid(0) + " idx=" + .@sIdx[.@i] + " id=" + .@sId[.@i] + " qty=" + .@sAmt[.@i];
		.@deleteOK = delitemidx(.@sIdx[.@i], .@sAmt[.@i]);
		if (.@deleteOK != 1) {
			logmes "WAKIM SELL INCIDENT delete_failed cid=" + getcharid(0) + " idx=" + .@sIdx[.@i] + " id=" + .@sId[.@i] + " qty=" + .@sAmt[.@i];
			@WakimOperationBusy = 0;
			mes "[Wakim]";
			mes "^FF0000Incomplete sale. GM recovery required.^000000";
			close;
		}
		// Verify the just-deleted inventory INDEX is no longer occupied.
		getinventorylist;
		.@still = 0;
		for (.@j=0; .@j < @inventorylist_count; ++.@j)
			if (@inventorylist_idx[.@j] == .@sIdx[.@i]) .@still = 1;
		if (.@still) {
			logmes "WAKIM SELL INCIDENT deletion_unverified cid=" + getcharid(0) + " idx=" + .@sIdx[.@i];
			@WakimOperationBusy = 0;
			mes "[Wakim]"; mes "^FF0000Incomplete sale. Incident recorded. GM recovery required.^000000"; close;
		}
		WakimSellDeleted++;
		logmes "WAKIM SELL DELETE_OK cid=" + getcharid(0) + " id=" + .@sId[.@i] + " qty=" + .@sAmt[.@i];
	}
	.@payBefore = Zeny;
	logmes "WAKIM SELL PAY_ATTEMPT cid=" + getcharid(0) + " before=" + .@payBefore + " payout=" + .@total;
	Zeny += .@total;
	if (Zeny != .@payBefore + .@total) {
		logmes "WAKIM SELL INCIDENT payout_unverified cid=" + getcharid(0);
		@WakimOperationBusy = 0;
		mes "[Wakim]"; mes "^FF0000Payout not verified. GM recovery required.^000000"; close;
	}
	logmes "WAKIM SELL SUCCESS cid=" + getcharid(0) + " slots=" + .@n + " payout=" + .@total;
	WakimSellIncident = 0;
	WakimSellExpected = 0;
	WakimSellDeleted = 0;
	WakimSellEpoch = 0;
	@WakimOperationBusy = 0;
	mes "[Wakim]";
	mes "Sold " + .@units + " units. Received " + .@total + "z.";
	close;

L_Admin:
	if (getgmlevel() < 99) close;
	mes "[Wakim - Administration]";
	mes "GM only. Exclusion changes affect ALL players.";
	mes "Native NoSell IDs in this build: " + .NoSellCount;
	mes "Sell All GM test: " + ($GR_WakimSaleEnabled ? "ENABLED" : "DISABLED");
	.@action = select("View excluded IDs:Add excluded Item ID:Remove excluded Item ID:Initialize empty exclusions:Toggle GM selling:View my incident:Clear MY incident (GM test only):Cancel");
	if (.@action == 8) close;
	clear;
	if (.@action == 1) {
		mes "[Wakim - Exclusions]";
		for (.@j=0; .@j < $GR_WakimExcludeCount; ++.@j)
			mes "ID " + $GR_WakimExcludes[.@j];
		mes "Revision: " + $GR_WakimExcludeRev;
		close;
	}
	if (.@action == 6 || .@action == 7) {
		mes "[Wakim - Incident]";
		mes "Current char: " + getcharid(0);
		mes "Flag: " + WakimSellIncident + ", attempted: " + WakimSellExpected + ", deleted: " + WakimSellDeleted;
		if (.@action == 6 || !WakimSellIncident) close;
		mes "^FF0000Only clear after reviewing logs and inventory.^000000";
		input .@reason$;
		if (getstrlen(.@reason$) < 10 || getgmlevel() < 99) close;
		if (select("CONFIRM MANUAL INCIDENT CLEAR:Cancel") != 1) close;
		if (getgmlevel() < 99 || !WakimSellIncident) close;
		logmes "WAKIM INCIDENT GM_CLEAR_SELF cid=" + getcharid(0) + " reason=" + .@reason$;
		WakimSellIncident = 0;
		WakimSellExpected = 0;
		WakimSellDeleted = 0;
		WakimSellEpoch = 0;
		mes "[Wakim]"; mes "Cleared current GM character incident."; close;
	}
	if (.@action == 4) {
		if ($GR_WakimExcludeReady) { mes "[Wakim]"; mes "Already initialized."; close; }
		if (select("Initialize empty exclusions:Cancel") != 1) close;
		if (getgmlevel() < 99) close;
		$GR_WakimExcludeCount = 0;
		$GR_WakimExcludeSum = 1;
		$GR_WakimExcludeRev++;
		$GR_WakimExcludeReady = 1;
		logmes "WAKIM ADMIN initialized exclusions gm=" + getcharid(0);
		mes "[Wakim]"; mes "Exclusions initialized."; close;
	}
	if (.@action == 5) {
		mes "[Wakim]";
		mes "^FF0000Real item deletion will be enabled for GMs.^000000";
		mes "Verify the registry, test case, and server backup first.";
		if (select("Toggle GM test:Cancel") != 1) close;
		if (getgmlevel() < 99 || !.NoSellReady || !$GR_WakimExcludeReady) close;
		$GR_WakimSaleEnabled = !$GR_WakimSaleEnabled;
		logmes "WAKIM ADMIN gm=" + getcharid(0) + " test_sale_enabled=" + $GR_WakimSaleEnabled;
		mes "[Wakim]"; mes "GM selling: " + ($GR_WakimSaleEnabled ? "ENABLED" : "DISABLED"); close;
	}
	if (.@action != 2 && .@action != 3) close;
	.@revision = $GR_WakimExcludeRev;
	mes "[Wakim]"; mes "Enter an Item ID.";
	input .@id;
	if (.@id <= 0 || .@id > 1000000) close;
	mes "[Wakim]"; mes "Item: " + .@id;
	if (select("Confirm exclusion edit:Cancel") != 1) close;
	if (getgmlevel() < 99 || $GR_WakimExcludeRev != .@revision || @WakimOperationBusy) {
		mes "[Wakim]"; mes "Stale request or current operation; canceled."; close;
	}
	if ($GR_WakimEditBusy && gettimetick(2) - $GR_WakimEditBusy <= 20) {
		mes "[Wakim]"; mes "Admin edit in progress."; close;
	}
	$GR_WakimEditBusy = gettimetick(2);
	.@pos = -1;
	for (.@i=0; .@i < $GR_WakimExcludeCount; ++.@i)
		if ($GR_WakimExcludes[.@i] == .@id) .@pos = .@i;
	if (.@action == 2 && .@pos == -1 && $GR_WakimExcludeCount < 40) {
		$GR_WakimExcludes[$GR_WakimExcludeCount] = .@id;
		$GR_WakimExcludeCount++;
	} else if (.@action == 3 && .@pos >= 0) {
		$GR_WakimExcludes[.@pos] = $GR_WakimExcludes[$GR_WakimExcludeCount-1];
		$GR_WakimExcludeCount--;
		$GR_WakimExcludes[$GR_WakimExcludeCount] = 0;
	} else {
		$GR_WakimEditBusy = 0;
		mes "[Wakim]"; mes "No change (duplicate/missing ID or list full)."; close;
	}
	$GR_WakimExcludeSum = 1;
	for (.@i=0; .@i < $GR_WakimExcludeCount; ++.@i)
		$GR_WakimExcludeSum += $GR_WakimExcludes[.@i] * (.@i + 1);
	$GR_WakimExcludeRev++;
	logmes "WAKIM ADMIN EXCLUSION gm=" + getcharid(0) + " id=" + .@id + " action=" + .@action + " rev=" + $GR_WakimExcludeRev;
	$GR_WakimEditBusy = 0;
	mes "[Wakim]"; mes "Exclusion revision: " + $GR_WakimExcludeRev; close;

L_Identify:
	// No temporary lock while the player is reading a dialog.
	if (@WakimOperationBusy) {
		mes "[Wakim]";
		mes "Another Wakim operation is already running.";
		close;
	}

	getinventorylist;
	.@slotCount = @inventorylist_count;
	.@units = 0;
	.@unidentifiedSlots = 0;
	for (.@i = 0; .@i < .@slotCount; .@i++) {
		// Snapshot all inventory entries, not just unidentified ones.
		.@ids[.@i] = @inventorylist_id[.@i];
		.@amounts[.@i] = @inventorylist_amount[.@i];
		.@identified[.@i] = @inventorylist_identify[.@i];
		if (.@identified[.@i] != 1) {
			.@units += .@amounts[.@i];
			.@unidentifiedSlots++;
		}
	}

	if (.@units == 0) {
		mes "[Wakim]";
		mes "You have no unidentified items.";
		close;
	}

	.@fee = (.@units <= 10) ? .@units * 40 : 0;
	mes "[Wakim]";
	mes "Unidentified units: " + .@units;
	mes "Unidentified inventory slots: " + .@unidentifiedSlots;
	mes "Identification fee: " + .@fee + " Zeny.";
	mes "11 or more units are identified for free.";
	next;
	if (select("Identify All:Cancel") != 1)
		close;

	// No player-input/yield commands from this point until completion.
	if (@WakimOperationBusy) {
		mes "[Wakim]";
		mes "Another Wakim operation started. Please try again.";
		close;
	}
	@WakimOperationBusy = 1;

	// Recalculate and verify the earlier snapshot, since 'next' and
	// 'select' allow time for the inventory to change.
	getinventorylist;
	if (@inventorylist_count != .@slotCount) {
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "Your inventory changed. Please try again.";
		close;
	}
	.@checkUnits = 0;
	.@checkSlots = 0;
	.@changed = 0;
	for (.@i = 0; .@i < .@slotCount; .@i++) {
		if (@inventorylist_id[.@i] != .@ids[.@i] ||
			@inventorylist_amount[.@i] != .@amounts[.@i] ||
			@inventorylist_identify[.@i] != .@identified[.@i]) {
			.@changed = 1;
			break;
		}
		if (@inventorylist_identify[.@i] != 1) {
			.@checkUnits += @inventorylist_amount[.@i];
			.@checkSlots++;
		}
	}
	if (.@changed || .@checkUnits != .@units || .@checkSlots != .@unidentifiedSlots) {
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "Your inventory changed. Please try again.";
		close;
	}
	if (Zeny < .@fee) {
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "You need " + .@fee + " Zeny to identify these items.";
		close;
	}
	if (identifyall(0) != .@unidentifiedSlots) {
		@WakimOperationBusy = 0;
		logmes "WAKIM IDENTIFY cancel: native slot count mismatch; cid=" + getcharid(0);
		mes "[Wakim]";
		mes "Identification state changed. Please retry.";
		close;
	}

	// identifyall identifies by inventory SLOT, not quantity.
	.@identifiedNow = identifyall(1);
	if (.@identifiedNow != .@unidentifiedSlots || identifyall(0) != 0) {
		logmes "WAKIM IDENTIFY incomplete; cid=" + getcharid(0) + "; expected_slots=" + .@unidentifiedSlots + "; actual_slots=" + .@identifiedNow;
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "Identification could not be fully verified. No fee was charged.";
		close;
	}

	if (.@fee > 0) {
		// The attempt must be recorded before the assignment: on a failed
		// SP_ZENY assignment rAthena can terminate the NPC script.
		.@zenyBefore = Zeny;
		if (.@zenyBefore < .@fee) {
			logmes "WAKIM IDENTIFY payment skipped after success (funds); cid=" + getcharid(0) + "; fee=" + .@fee;
			@WakimOperationBusy = 0;
			mes "[Wakim]";
			mes "Identification completed, but no fee could be collected. An audit was recorded.";
			close;
		}
		logmes "WAKIM IDENTIFY payment_attempt; cid=" + getcharid(0) + "; units=" + .@units + "; fee=" + .@fee + "; before=" + .@zenyBefore;
		Zeny -= .@fee;
		if (Zeny != .@zenyBefore - .@fee) {
			logmes "WAKIM IDENTIFY payment_unverified; cid=" + getcharid(0) + "; fee=" + .@fee;
			@WakimOperationBusy = 0;
			mes "[Wakim]";
			mes "Identification completed, but payment requires review.";
			close;
		}
	}

	logmes "WAKIM IDENTIFY success; cid=" + getcharid(0) + "; units=" + .@units + "; fee=" + .@fee;
	@WakimOperationBusy = 0;
	mes "[Wakim]";
	mes "Successfully identified " + .@units + " item units.";
	mes "Paid: " + .@fee + " Zeny.";
	close;
