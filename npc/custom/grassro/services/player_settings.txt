
//==========================================================
// Player Settings NPC v2 - rAthena
// Account-wide persistent settings
// Prontera: 155,180
//==========================================================

prontera,164,175,2	script	Player Settings	837,{

Main:
	mes "[Player Settings]";
	mes "Hello, "+strcharinfo(0)+"!";
	mes "All preferences are shared by your account.";
	mes " ";
	mes "AutoLoot: "+(#PS_Loot ? "ON ("+#PS_Loot+"%)" : "OFF");
	mes "Show EXP: "+(#PS_EXP ? "ON" : "OFF");
	mes "Show Zeny: "+(#PS_Zeny ? "ON" : "OFF");
	mes "Show Delay: "+(#PS_Delay ? "ON" : "OFF");
	mes "NoAsk: "+(#PS_NoAsk ? "ON" : "OFF");
	next;

	switch(select(
		"AutoLoot Settings:"+
		"Display Settings:"+
		"Login Settings:"+
		"Social Settings:"+
		"Reset All Settings:"+
		"Exit"
	)) {
		case 1: goto LootMenu;
		case 2: goto DisplayMenu;
		case 3: goto LoginMenu;
		case 4: goto SocialMenu;
		case 5: goto ResetMenu;
		default: close;
	}

//==========================================================
// AUTOLOOT
//==========================================================

LootMenu:
	mes "[AutoLoot Settings]";
	mes "Current: "+(#PS_Loot ? #PS_Loot+"%" : "OFF");
	mes "Saved Item IDs: "+#PS_Item1+", "+#PS_Item2+", "+#PS_Item3;
	mes "Item Type: "+(#PS_Type ? "Enabled" : "Disabled");
	next;

	switch(select(
		"Toggle AutoLoot:"+
		"Set AutoLoot Percentage:"+
		"Add AutoLoot Item ID:"+
		"Clear AutoLoot Item IDs:"+
		"Set AutoLoot Item Type:"+
		"Clear AutoLoot Item Type:"+
		"Back"
	)) {
		case 1:
			if (#PS_Loot) {
				#PS_Loot = 0;
				atcommand "@autoloot off";
			} else {
				#PS_Loot = 100;
				atcommand "@autoloot 100";
			}
			dispbottom "AutoLoot updated.";
			goto LootMenu;

		case 2:
			mes "[AutoLoot]";
			mes "Enter a percentage from 1 to 100.";
			mes "Only items with drop rates at or below";
			mes "this percentage will be autolooted.";
			next;
			input .@rate;
			if (.@rate < 1 || .@rate > 100) {
				mes "Invalid percentage.";
				close;
			}
			#PS_Loot = .@rate;
			atcommand "@autoloot "+#PS_Loot;
			dispbottom "AutoLoot set to "+#PS_Loot+"%.";
			close;

		case 3:
			mes "[AutoLoot Item ID]";
			mes "Enter the numeric item ID.";
			mes "Maximum: 3 saved IDs.";
			next;
			input .@id;
			if (.@id <= 0 || .@id > 1000000) {
				mes "Invalid item ID.";
				close;
			}
			if (!getitemname(.@id)) {
				mes "Item not found.";
				close;
			}
			if (#PS_Item1 == .@id ||
			    #PS_Item2 == .@id ||
			    #PS_Item3 == .@id) {
				mes "This item is already saved.";
				close;
			}
			if (!#PS_Item1)
				#PS_Item1 = .@id;
			else if (!#PS_Item2)
				#PS_Item2 = .@id;
			else if (!#PS_Item3)
				#PS_Item3 = .@id;
			else {
				mes "All three slots are full.";
				close;
			}
			atcommand "@alootid +"+.@id;
			dispbottom "Saved AutoLoot item: "+.@id;
			close;

		case 4:
			#PS_Item1 = 0;
			#PS_Item2 = 0;
			#PS_Item3 = 0;
			atcommand "@alootid reset";
			dispbottom "AutoLoot item IDs cleared.";
			close;

		case 5:
			mes "[AutoLoot Item Type]";
			mes "Select the category to AutoLoot.";
			next;
			.@choice = select(
				"Healing Items:"+
				"Usable Items:"+
				"Etc Items:"+
				"Weapons:"+
				"Armor:"+
				"Cards:"+
				"Pet Eggs:"+
				"Pet Armor:"+
				"Ammunition:"+
				"Cancel"
			);
			if (.@choice == 10) close;
			switch(.@choice) {
				case 1: .@type = 0; break;
				case 2: .@type = 2; break;
				case 3: .@type = 3; break;
				case 4: .@type = 4; break;
				case 5: .@type = 5; break;
				case 6: .@type = 6; break;
				case 7: .@type = 7; break;
				case 8: .@type = 8; break;
				case 9: .@type = 10; break;
			}
			#PS_Type = .@type + 1;
			atcommand "@autoloottype reset";
			atcommand "@autoloottype +"+.@type;
			dispbottom "AutoLoot item type updated.";
			close;

		case 6:
			#PS_Type = 0;
			atcommand "@autoloottype reset";
			dispbottom "AutoLoot item type cleared.";
			close;

		default: goto Main;
	}

//==========================================================
// DISPLAY SETTINGS
//==========================================================

DisplayMenu:
	mes "[Display Settings]";
	mes "Show EXP: "+(#PS_EXP ? "ON" : "OFF");
	mes "Show Zeny: "+(#PS_Zeny ? "ON" : "OFF");
	mes "Show Delay: "+(#PS_Delay ? "ON" : "OFF");
	next;

	switch(select(
		"Toggle Show EXP:"+
		"Toggle Show Zeny:"+
		"Toggle Show Delay:"+
		"Back"
	)) {
		case 1:
			#PS_EXP = !#PS_EXP;
			atcommand "@showexp";
			break;
		case 2:
			#PS_Zeny = !#PS_Zeny;
			atcommand "@showzeny";
			break;
		case 3:
			#PS_Delay = !#PS_Delay;
			atcommand "@showdelay";
			break;
		default: goto Main;
	}
	dispbottom "Display preference updated.";
	goto DisplayMenu;

//==========================================================
// LOGIN SETTINGS
//==========================================================

LoginMenu:
	mes "[Login Settings]";
	mes "Show Rates: "+(#PS_Rates ? "ON" : "OFF");
	mes "Show Uptime: "+(#PS_Uptime ? "ON" : "OFF");
	mes " ";
	mes "These commands run automatically at login.";
	next;

	switch(select(
		"Toggle Show Rates on Login:"+
		"Toggle Show Uptime on Login:"+
		"Show Rates Now:"+
		"Show Uptime Now:"+
		"Back"
	)) {
		case 1:
			#PS_Rates = !#PS_Rates;
			break;
		case 2:
			#PS_Uptime = !#PS_Uptime;
			break;
		case 3:
			atcommand "@rates";
			break;
		case 4:
			atcommand "@uptime";
			break;
		default: goto Main;
	}
	goto LoginMenu;

//==========================================================
// SOCIAL SETTINGS
//==========================================================

SocialMenu:
	mes "[Social Settings]";
	mes "NoAsk: "+(#PS_NoAsk ? "ON" : "OFF");
	mes "Main Chat on Login: "+(#PS_Main ? "ON" : "OFF");
	mes " ";
	mes "NoAsk automatically rejects invitations.";
	next;

	switch(select(
		"Toggle NoAsk:"+
		"Toggle Main Chat on Login:"+
		"Back"
	)) {
		case 1:
			#PS_NoAsk = !#PS_NoAsk;
			atcommand "@noask";
			goto SocialMenu;

		case 2:
			#PS_Main = !#PS_Main;
			if (#PS_Main)
				atcommand "@main on";
			else
				atcommand "@main off";
			goto SocialMenu;

		default: goto Main;
	}

//==========================================================
// RESET ALL SETTINGS
//==========================================================

ResetMenu:
	mes "[Player Settings]";
	mes "Reset all account-wide preferences?";
	mes "This affects every character on your account.";
	next;

	if (select("Yes, reset:Cancel") == 2)
		goto Main;

	if (#PS_EXP) atcommand "@showexp";
	if (#PS_Zeny) atcommand "@showzeny";
	if (#PS_Delay) atcommand "@showdelay";
	if (#PS_NoAsk) atcommand "@noask";
	if (#PS_Main) atcommand "@main off";

	atcommand "@autoloot off";
	atcommand "@alootid reset";
	atcommand "@autoloottype reset";

	#PS_Loot = 0;
	#PS_Item1 = 0;
	#PS_Item2 = 0;
	#PS_Item3 = 0;
	#PS_Type = 0;
	#PS_EXP = 0;
	#PS_Zeny = 0;
	#PS_Delay = 0;
	#PS_NoAsk = 0;
	#PS_Main = 0;
	#PS_Rates = 0;
	#PS_Uptime = 0;

	mes "[Player Settings]";
	mes "All saved preferences have been reset.";
	close;
}

//==========================================================
// ACCOUNT SETTINGS RESTORATION
// Invisible login event NPC
//==========================================================

-	script	PlayerSettingsLogin	-1,{

OnPCLoginEvent:
	// Allow character login initialization to finish.
	sleep2 1000;

	// Clear session-based AutoLoot selections.
	atcommand "@autoloot off";
	atcommand "@alootid reset";
	atcommand "@autoloottype reset";

	// Restore AutoLoot percentage.
	if (#PS_Loot > 0)
		atcommand "@autoloot "+#PS_Loot;

	// Restore saved individual item IDs.
	if (#PS_Item1 > 0)
		atcommand "@alootid +"+#PS_Item1;
	if (#PS_Item2 > 0)
		atcommand "@alootid +"+#PS_Item2;
	if (#PS_Item3 > 0)
		atcommand "@alootid +"+#PS_Item3;

	// Restore selected item type.
	if (#PS_Type > 0)
		atcommand "@autoloottype +"+(#PS_Type - 1);

	// Restore display preferences.
	if (#PS_EXP)
		atcommand "@showexp";
	if (#PS_Zeny)
		atcommand "@showzeny";
	if (#PS_Delay)
		atcommand "@showdelay";

	// Restore social preferences.
	if (#PS_NoAsk)
		atcommand "@noask";
	if (#PS_Main)
		atcommand "@main on";

	// Optional login information.
	if (#PS_Rates)
		atcommand "@rates";
	if (#PS_Uptime)
		atcommand "@uptime";

	end;
}
