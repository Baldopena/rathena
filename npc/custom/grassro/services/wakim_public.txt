		@WakimOperationBusy = 0;
		mes "[Wakim]"; mes "Inventory changed during SQL preparation. No sale; ledger locked."; close;

	// Never release this SQL guard until a later login has verified persisted state.

		getinventorylist;
		.@still = 0;
		for (.@j=0; .@j<@inventorylist_count; ++.@j)
			if (@inventorylist_idx[.@j] == .@sIdx[.@i]) .@still = 1;
		if (.@still) { @WakimOperationBusy = 0; mes "[Wakim]"; mes "Deletion could not be verified. SQL incident locked."; close; }

		.@check = 0;
		.@sqlRows = query_sql("SELECT COUNT(*) FROM rathena.wakim_sale_event WHERE txn_id=" + .@txid + " AND event_seq=" + (.@seq+1) + " AND kind='DELETE_OBSERVED'", .@check);
		if (.@sqlRows != 1 || .@check != 1) { @WakimOperationBusy = 0; mes "[Wakim]"; mes "Deletion audit incomplete. SQL incident locked."; close; }

// Only this successful-path reconciliation can release the active SQL guard.

S_ReconcileSavedSale:
	.@rcCid = getcharid(0);
	.@rcTid = 0; .@rcSlots = 0; .@rcDeleted = 0;
	.@rcInitial = 0; .@rcAfter = 0; .@rcPayout = 0; .@rcSaved = 0;
	.@rcUpdated = 0; .@rcStatus$ = "";
	.@rcRows = query_sql("SELECT txn_id,status,expected_slots,observed_deleted_slots,initial_zeny,observed_zeny_after,payout_zeny,item_rows_recorded,UNIX_TIMESTAMP(updated_at) FROM rathena.wakim_sale_txn WHERE active_char_guard=" + .@rcCid, .@rcTid,.@rcStatus$,.@rcSlots,.@rcDeleted,.@rcInitial,.@rcAfter,.@rcPayout,.@rcSaved,.@rcUpdated);
	if (.@rcRows != 1 || .@rcStatus$ != "NEEDS_REVIEW" || .@rcTid <= 0) { return; }
	if (@WakimSaleLoginTick <= .@rcUpdated || .@rcSlots < 1 || .@rcSlots > 100 || .@rcDeleted != .@rcSlots || .@rcSaved != .@rcSlots || .@rcInitial + .@rcPayout != .@rcAfter || Zeny != .@rcAfter) { return; }
	if (WakimSellIncident != 1 || WakimSellExpected != .@rcSlots || WakimSellDeleted != .@rcSlots) { return; }
	.@rcIntents = 0; .@rcObserved = 0; .@rcPayIntents = 0; .@rcPayObserved = 0;
	.@rcRows = query_sql("SELECT COALESCE(SUM(kind='DELETE_INTENT'),0),COALESCE(SUM(kind='DELETE_OBSERVED'),0),COALESCE(SUM(kind='PAYMENT_INTENT'),0),COALESCE(SUM(kind='PAYMENT_OBSERVED'),0) FROM rathena.wakim_sale_event WHERE txn_id=" + .@rcTid, .@rcIntents,.@rcObserved,.@rcPayIntents,.@rcPayObserved);
	if (.@rcRows != 1 || .@rcIntents != .@rcSlots || .@rcObserved != .@rcSlots || .@rcPayIntents != 1 || .@rcPayObserved != 1) { return; }
	.@rcRows = query_sql("SELECT item_id,quantity,COALESCE(CAST(JSON_UNQUOTE(JSON_EXTRACT(attributes_json,'$.inventory_id_total_before')) AS UNSIGNED),0) FROM rathena.wakim_sale_item WHERE txn_id=" + .@rcTid + " ORDER BY line_no", .@rcItemId,.@rcQuantity,.@rcBefore);
	if (.@rcRows != .@rcSlots) { return; }
	getinventorylist;
	for (.@rcI=0; .@rcI<.@rcSlots; ++.@rcI) {
		.@rcSeen = 0;
		for (.@rcK=0; .@rcK<.@rcI; ++.@rcK)
			if (.@rcItemId[.@rcK] == .@rcItemId[.@rcI]) .@rcSeen = 1;
		if (.@rcSeen) continue;
		.@rcSold = 0;
		for (.@rcK=0; .@rcK<.@rcSlots; ++.@rcK)
			if (.@rcItemId[.@rcK] == .@rcItemId[.@rcI]) .@rcSold += .@rcQuantity[.@rcK];
		.@rcBeforeTotal = .@rcBefore[.@rcI];
		// Legacy GM pilot row had 28 fields, so require ZERO remaining
		// copies of its sold ID instead of guessing the earlier count.
		if (.@rcBeforeTotal == 0) .@rcBeforeTotal = .@rcSold;
		if (.@rcBeforeTotal < .@rcSold) { return; }
		.@rcNow = 0;
		for (.@rcJ=0; .@rcJ<@inventorylist_count; ++.@rcJ)
			if (@inventorylist_id[.@rcJ] == .@rcItemId[.@rcI]) .@rcNow += @inventorylist_amount[.@rcJ];
		if (.@rcNow != .@rcBeforeTotal - .@rcSold) { return; }
	}
	// Record proof BEFORE releasing SQL guard; retry safely on interrupted write.
	.@rcSeq = .@rcSlots * 2 + 4;
	.@rcCheck = 0;
	.@rcRows = query_sql("SELECT COUNT(*) FROM rathena.wakim_sale_event WHERE txn_id=" + .@rcTid + " AND event_seq=" + .@rcSeq + " AND kind='SAVE_RECONCILED'", .@rcCheck);
	if (.@rcRows != 1) { return; }
	if (.@rcCheck == 0)
		query_sql("INSERT INTO rathena.wakim_sale_event(txn_id,event_seq,kind,actor_account_id,actor_char_id,detail_json) VALUES (" + .@rcTid + "," + .@rcSeq + ",'SAVE_RECONCILED'," + getcharid(3) + "," + .@rcCid + ",JSON_OBJECT('saved_zeny'," + Zeny + ",'login_tick'," + @WakimSaleLoginTick + "))");
	.@rcCheck = 0;
	.@rcRows = query_sql("SELECT COUNT(*) FROM rathena.wakim_sale_event WHERE txn_id=" + .@rcTid + " AND event_seq=" + .@rcSeq + " AND kind='SAVE_RECONCILED'", .@rcCheck);
	if (.@rcRows != 1 || .@rcCheck != 1) { return; }
	query_sql("UPDATE rathena.wakim_sale_txn SET status='COMPLETED',completed_at=NOW(6) WHERE txn_id=" + .@rcTid + " AND status='NEEDS_REVIEW' AND char_id=" + .@rcCid + " AND observed_deleted_slots=expected_slots AND item_rows_recorded=expected_slots AND observed_zeny_after=" + Zeny);
	.@rcStatus$ = "";
	.@rcRows = query_sql("SELECT status FROM rathena.wakim_sale_txn WHERE txn_id=" + .@rcTid + " AND char_id=" + .@rcCid, .@rcStatus$);
	if (.@rcRows != 1 || .@rcStatus$ != "COMPLETED") { return; }
	WakimSellIncident = 0;
	WakimSellExpected = 0;
	WakimSellDeleted = 0;
	WakimSellEpoch = 0;
	logmes "WAKIM SAVE_RECONCILED txid=" + .@rcTid + " cid=" + .@rcCid;
	return;

		.@sqlRows = query_sql("SELECT COUNT(*) FROM rathena.wakim_sale_txn t WHERE t.active_char_guard IS NOT NULL AND (t.status<>'NEEDS_REVIEW' OR t.observed_deleted_slots<>t.expected_slots OR t.observed_zeny_after IS NULL OR NOT EXISTS (SELECT 1 FROM rathena.wakim_sale_event e WHERE e.txn_id=t.txn_id AND e.kind='PAYMENT_OBSERVED'))", .@openCount);

L_Identify:

	if (@WakimOperationBusy) {
		mes "[Wakim]";
		mes "Another Wakim operation is already running.";
		close;
	}

	getinventorylist;
	.@slotCount = @inventorylist_count;
	.@units = 0;
	.@unidentifiedSlots = 0;
	for (.@i = 0; .@i < .@slotCount; .@i++) {
		// Snapshot all inventory entries, not just unidentified ones.
		.@ids[.@i] = @inventorylist_id[.@i];
		.@amounts[.@i] = @inventorylist_amount[.@i];
		.@identified[.@i] = @inventorylist_identify[.@i];
		if (.@identified[.@i] != 1) {
			.@units += .@amounts[.@i];
			.@unidentifiedSlots++;
		}
	}

	if (.@units == 0) {
		mes "[Wakim]";
		mes "You have no unidentified items.";
		close;
	}

	.@fee = (.@units <= 10) ? .@units * 40 : 0;
	mes "[Wakim]";
	mes "Unidentified units: " + .@units;
	mes "Unidentified inventory slots: " + .@unidentifiedSlots;
	mes "Identification fee: " + .@fee + " Zeny.";
	mes "11 or more units are identified for free.";
	next;
	if (select("Identify All:Cancel") != 1)
		close;

	// No player-input/yield commands from this point until completion.
	if (@WakimOperationBusy) {
		mes "[Wakim]";
		mes "Another Wakim operation started. Please try again.";
		close;
	}
	@WakimOperationBusy = 1;

	// Recalculate and verify the earlier snapshot, since 'next' and
	// 'select' allow time for the inventory to change.
	getinventorylist;
	if (@inventorylist_count != .@slotCount) {
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "Your inventory changed. Please try again.";
		close;
	}
	.@checkUnits = 0;
	.@checkSlots = 0;
	.@changed = 0;
	for (.@i = 0; .@i < .@slotCount; .@i++) {
		if (@inventorylist_id[.@i] != .@ids[.@i] ||
			@inventorylist_amount[.@i] != .@amounts[.@i] ||
			@inventorylist_identify[.@i] != .@identified[.@i]) {
			.@changed = 1;
			break;
		}
		if (@inventorylist_identify[.@i] != 1) {
			.@checkUnits += @inventorylist_amount[.@i];
			.@checkSlots++;
		}
	}
	if (.@changed || .@checkUnits != .@units || .@checkSlots != .@unidentifiedSlots) {
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "Your inventory changed. Please try again.";
		close;
	}
	if (Zeny < .@fee) {
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "You need " + .@fee + " Zeny to identify these items.";
		close;
	}
	if (identifyall(0) != .@unidentifiedSlots) {
		@WakimOperationBusy = 0;
		logmes "WAKIM IDENTIFY cancel: native slot count mismatch; cid=" + getcharid(0);
		mes "[Wakim]";
		mes "Identification state changed. Please retry.";
		close;
	}

	// identifyall identifies by inventory SLOT, not quantity.
	.@identifiedNow = identifyall(1);
	if (.@identifiedNow != .@unidentifiedSlots || identifyall(0) != 0) {
		logmes "WAKIM IDENTIFY incomplete; cid=" + getcharid(0) + "; expected_slots=" + .@unidentifiedSlots + "; actual_slots=" + .@identifiedNow;
		@WakimOperationBusy = 0;
		mes "[Wakim]";
		mes "Identification could not be fully verified. No fee was charged.";
		close;
	}

	if (.@fee > 0) {
		// The attempt must be recorded before the assignment: on a failed
		// SP_ZENY assignment rAthena can terminate the NPC script.
		.@zenyBefore = Zeny;
		if (.@zenyBefore < .@fee) {
			logmes "WAKIM IDENTIFY payment skipped after success (funds); cid=" + getcharid(0) + "; fee=" + .@fee;
			@WakimOperationBusy = 0;
			mes "[Wakim]";
			mes "Identification completed, but no fee could be collected. An audit was recorded.";
			close;
		}
		logmes "WAKIM IDENTIFY payment_attempt; cid=" + getcharid(0) + "; units=" + .@units + "; fee=" + .@fee + "; before=" + .@zenyBefore;
		Zeny -= .@fee;
		if (Zeny != .@zenyBefore - .@fee) {
			logmes "WAKIM IDENTIFY payment_unverified; cid=" + getcharid(0) + "; fee=" + .@fee;
			@WakimOperationBusy = 0;
			mes "[Wakim]";
			mes "Identification completed, but payment requires review.";
			close;
		}
	}

	logmes "WAKIM IDENTIFY success; cid=" + getcharid(0) + "; units=" + .@units + "; fee=" + .@fee;
	@WakimOperationBusy = 0;
	mes "[Wakim]";
	mes "Successfully identified " + .@units + " item units.";
	mes "Paid: " + .@fee + " Zeny.";
	close;
