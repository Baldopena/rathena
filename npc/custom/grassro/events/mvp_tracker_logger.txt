OnNaturalMVP:
	if (killedrid != .@nat_mob || strcharinfo(3) != .@nat_map$)
		end;

	// Preserve the existing MVP rewards and announcements.
	doevent "RagnarokCoinDrop::OnNPCKillEvent";
	doevent "GR_MVPVoucher::OnNPCKillEvent";
	doevent "GR_MVP_Kill_Announcement::OnNPCKillEvent";

	// Verify the natural spawn against the registry.
	.@origin$ = "unknown";
	.@spawn_id = 0;

	.@match = query_sql(
		"SELECT `id` FROM `grassro_mvp_spawns` " +
		"WHERE `id`=" + .@nat_id +
		" AND `mob_id`=" + .@nat_mob +
		" AND `map_name`='" + escape_sql(.@nat_map$) + "'" +
		" AND `source_file`='" + escape_sql(.@nat_file$) + "'" +
		" AND `enabled`=1 LIMIT 1",
		.@found_id
	);

	if (.@match == 1) {
		.@spawn_id = .@found_id;
		.@origin$ = "natural";
	} else {
		logmes "[GR_MVP_TRACKER] Natural spawn registry mismatch: " + .@nat_id;
	}

	goto OnRecordKill;

OnRecordKill:
	.@mob_id = killedrid;

	// Ignore invalid monster IDs and non-MVP monsters.
	if (.@mob_id <= 0)
		end;

	if (getmonsterinfo(.@mob_id, MOB_MVPEXP) <= 0)
		end;

	// Identify credited character.
	.@char_id = getcharid(0);
	.@account_id = getcharid(3);

	if (.@char_id <= 0 || .@account_id <= 0)
		end;

	// Collect kill information.
	.@char_name$ = escape_sql(strcharinfo(0));
	.@mob_name$ = escape_sql(getmonsterinfo(.@mob_id, MOB_NAME));
	.@map_name$ = escape_sql(strcharinfo(3));

	.@spawn_sql$ = "NULL";
	if (.@spawn_id > 0)
		.@spawn_sql$ = "" + .@spawn_id;

	// Record kill to permanent database.
	query_sql(
		"INSERT INTO `grassro_mvp_kills` " +
		"(`account_id`, `char_id`, `char_name`, " +
		"`mob_id`, `mob_name`, `map_name`, " +
		"`spawn_id`, `kill_source`, `killed_at`) VALUES (" +
		.@account_id + "," +
		.@char_id + ",'" +
		.@char_name$ + "'," +
		.@mob_id + ",'" +
		.@mob_name$ + "','" +
		.@map_name$ + "'," +
		.@spawn_sql$ + ",'" +
		.@origin$ + "', UTC_TIMESTAMP())"
	);

	end;
