// GrassRO Costume Converter - generated mappings, independently configurable fee.
// Players may only convert unequipped, unbound, non-rental eligible headgear.
// All original cards, refinement, enchant options and effects are destroyed.
prontera,155,180,4	script	GrassRO Costume Converter	4_F_KAFRA1,{
    if (getgmlevel() >= .AdminLevel) {
        mes "[Costume Converter]";
        .@a = select("Convert headgear:GM administration:Cancel");
        if (.@a == 2) goto L_Admin;
        if (.@a == 3) close;
    }
    if (!$GR_CC_Enabled) {
        mes "[Costume Converter]";
        mes "Costume conversion is currently unavailable.";
        close;
    }
    getinventorylist;
    .@choices$ = "";
    .@count = 0;
    for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
        .@id = @inventorylist_id[.@i];
        if (!.CostumeMap[.@id]) continue;
        if (@inventorylist_equip[.@i]) continue;
        if (@inventorylist_bound[.@i] || @inventorylist_expire[.@i]) continue;
        .@excluded = getd("$GR_CC_Exclude_"+.@id);
        if (.@excluded == -1 || .@excluded > gettimetick(2)) continue;
        .@slots[.@count] = @inventorylist_idx[.@i];
        .@ids[.@count] = .@id;
        // Snapshot the selected exact instance to prevent a slot-swap surprise.
        .@refines[.@count] = @inventorylist_refine[.@i];
        .@cards1[.@count] = @inventorylist_card1[.@i];
        .@cards2[.@count] = @inventorylist_card2[.@i];
        .@cards3[.@count] = @inventorylist_card3[.@i];
        .@cards4[.@count] = @inventorylist_card4[.@i];
        .@choices$ += replacestr(getitemname(.@id),":"," ") + " [" + .@id + "] +" + @inventorylist_refine[.@i] + ":";
        .@count++;
        if (.@count >= 25) break; // Stay below legacy client menu length limits.
    }
    mes "[Costume Converter]";
    if (!.@count) {
        mes "You have no eligible unequipped headgear in your inventory.";
        close;
    }
    mes "Select a headgear. Conversion costs " + .FeeAmount + " " + getitemname(.FeeItem) + ".";
    if (.@count == 25) mes "Showing the first 25 eligible items only.";
    next;
    .@choices$ = substr(.@choices$,0,getstrlen(.@choices$)-2);
    .@choice = select(.@choices$) - 1;
    if (.@choice < 0 || .@choice >= .@count) close;
    .@idx = .@slots[.@choice];
    .@id = .@ids[.@choice];
    .@costume = .CostumeMap[.@id];
    .@refine = .@refines[.@choice];
    .@card1 = .@cards1[.@choice];
    .@card2 = .@cards2[.@choice];
    .@card3 = .@cards3[.@choice];
    .@card4 = .@cards4[.@choice];
    mes "[Costume Converter]";
    mes "Original: " + getitemname(.@id);
    mes "Result: " + getitemname(.@costume);
    mes "The original, cards, refinement and existing enchantments will be permanently lost.";
    mes "Cost: " + .FeeAmount + " " + getitemname(.FeeItem) + ".";
    next;
    if (select("Confirm permanent conversion:Cancel") != 1) close;
    if (!$GR_CC_Enabled || !.CostumeMap[.@id]) close;
    .@excluded = getd("$GR_CC_Exclude_"+.@id);
    if (.@excluded == -1 || .@excluded > gettimetick(2)) close;
    if (countitem(.FeeItem) < .FeeAmount) {
        mes "Not enough " + getitemname(.FeeItem) + ".";
        close;
    }
    if (!checkweight(.@costume,1)) {
        mes "Your inventory cannot accept the costume.";
        close;
    }
    // Refresh inventory indices after the confirmation dialog and verify exact slot.
    getinventorylist;
    .@found = 0;
    for (.@j = 0; .@j < @inventorylist_count; ++.@j) {
        if (@inventorylist_idx[.@j] != .@idx || @inventorylist_id[.@j] != .@id) continue;
        if (@inventorylist_equip[.@j] || @inventorylist_bound[.@j] || @inventorylist_expire[.@j]) break;
        if (@inventorylist_refine[.@j] != .@refine ||
            @inventorylist_card1[.@j] != .@card1 || @inventorylist_card2[.@j] != .@card2 ||
            @inventorylist_card3[.@j] != .@card3 || @inventorylist_card4[.@j] != .@card4) break;
        .@found = 1;
        break;
    }
    if (!.@found) {
        mes "The selected headgear changed. Please try again.";
        close;
    }
    if (!delitemidx(.@idx,1)) {
        mes "Unable to consume headgear. Nothing was charged.";
        close;
    }
    delitem .FeeItem,.FeeAmount;
    getitem .@costume,1;
    logmes "GrassRO Costume Converter: original="+.@id+" costume="+.@costume+" fee="+.FeeAmount;
    mes "Conversion completed.";
    close;

L_Admin:
    if (getgmlevel() < .AdminLevel) close;
    mes "[GM Costume Administration]";
    mes "System: " + ($GR_CC_Enabled ? "ENABLED" : "DISABLED");
    next;
    .@a = select("Enable/disable converter:Exclude item:Remove exclusion:View exclusions:Exit");
    if (.@a == 5) close;
    if (.@a == 1) {
        $GR_CC_Enabled = !$GR_CC_Enabled;
        mes "Converter is now " + ($GR_CC_Enabled ? "ENABLED." : "DISABLED.");
        close;
    }
    if (.@a == 4) {
        .@shown = 0;
        .@total = 0;
        for (.@i=0; .@i < .RegisteredCount; ++.@i) {
            .@id = .OriginalIds[.@i];
            .@until = getd("$GR_CC_Exclude_"+.@id);
            if (.@until != -1 && .@until <= gettimetick(2)) continue;
            .@total++;
            if (.@shown < 20) {
                mes .@id + " - " + getitemname(.@id) + " (" + (.@until == -1 ? "permanent" : "temporary") + ")";
                .@shown++;
            }
        }
        mes "Active exclusions: " + .@total + " (showing up to 20).";
        close;
    }
    mes "Enter original headgear item ID:";
    input .@id;
    if (.@id <= 0 || .@id > 65535 || !.CostumeMap[.@id]) {
        mes "Not in the current costume registry.";
        close;
    }
    if (.@a == 3) {
        setd "$GR_CC_Exclude_"+.@id,0;
        mes "Exclusion removed.";
        close;
    }
    next;
    .@b = select("Permanent exclusion:Temporary exclusion (days):Cancel");
    if (.@b == 3) close;
    if (.@b == 1) {
        setd "$GR_CC_Exclude_"+.@id,-1;
        mes "Permanent exclusion saved.";
        close;
    }
    mes "Number of days (1 to 3650):";
    input .@days;
    if (.@days < 1 || .@days > 3650) close;
    setd "$GR_CC_Exclude_"+.@id,gettimetick(2)+(.@days*86400);
    mes "Temporary exclusion saved.";
    close;
